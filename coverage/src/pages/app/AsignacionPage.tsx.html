/**

  const [error, setError] = useState<string | null>(null);

      const historial = await api.get<SolicitudHistorial[]>("/solicitudes/historial");

        setSinSolicitudes(true);
        return;

      setError(mensajeDeError(err, "No pudimos cargar tu resultado."));
    } finally {
      setCargando(false);
    }
